// LLMDropdownComponent.ts
populateLLMDropdown() {
  this.llmService.getAvailableLLMs().subscribe(llms => {
    // Filter out non-functional or invalid LLMs before populating dropdown
    this.llmOptions = llms.filter(llm => this.validateLLMConfiguration(llm));
  });
}

// Add validation method to check LLM operational status
validateLLMConfiguration(llm: LLM): boolean {
  // Example validation: check required fields and operational flag
  return llm.isEnabled && llm.endpointUrl !== '' && llm.status === 'operational';
}

// PromptExecutionService.ts
async executePromptWithSelectedLLM(selectedLLMId: string, prompt: string): Promise<string> {
  try {
    const llm = this.llmService.getLLMById(selectedLLMId);
    if (!this.validateLLMConfiguration(llm)) {
      throw new Error('Selected LLM is not functional or supported.');
    }
    await llm.loadModel(); // Await loading to catch errors
    return await llm.executePrompt(prompt);
  } catch (error) {
    // Handle errors gracefully and provide user-friendly message
    this.notificationService.showError(`Failed to execute prompt: ${error.message}`);
    throw error; // Optionally rethrow or handle as per app flow
  }
}

// Reuse the same validation method or import from a shared validator
validateLLMConfiguration(llm: LLM): boolean {
  return llm.isEnabled && llm.endpointUrl !== '' && llm.status === 'operational';
}
