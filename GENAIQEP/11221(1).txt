public ResponseEntity<?> uploadToJiraALM(String workItemId, TestData data) {
    // Validate workItemId before processing
    if (!WorkItemValidator.validateWorkItemId(workItemId)) {
        // Return structured error response with specific message
        return ResponseEntity.badRequest().body("Please provide the valid work item Id");
    }
    try {
        jiraAlmService.uploadData(workItemId, data);
        return ResponseEntity.ok("Upload successful");
    } catch (ValidationException ve) {
        // Catch specific validation exceptions and return meaningful message
        return ResponseEntity.badRequest().body(ve.getMessage());
    } catch (Exception e) {
        // Fallback generic error
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body("Error");
    }
}

public class WorkItemValidator {
    public static boolean validateWorkItemId(String workItemId) {
        // Example validation: non-null, non-empty, matches expected pattern (e.g., alphanumeric)
        if (workItemId == null || workItemId.trim().isEmpty()) {
            return false;
        }
        // Assuming work item IDs are alphanumeric strings of length 5-10
        return workItemId.matches("^[a-zA-Z0-9]{5,10}$");
    }
}
