async loadAvailableLLMs() {
  const allLLMs = await this.llmService.getAllLLMs();
  const validatedLLMs = [];

  for (const llm of allLLMs) {
    try {
      const isValid = await this.llmService.validateLLM(llm.id);
      if (isValid) {
        validatedLLMs.push(llm);
      }
    } catch (error) {
      console.warn(`LLM validation failed for ${llm.id}:`, error);
      // Optionally log or report error for monitoring
    }
  }

  this.dropdownOptions = validatedLLMs; // only include validated LLMs
}


Changes Made:  
- Added asynchronous validation of each LLM before adding to dropdown.  
- Filtered out non-functional or unsupported LLMs.  
- Added error handling during validation to avoid UI crashes.

async validateLLM(llmId: string): Promise<boolean> {
  try {
    const response = await apiClient.get(`/llm/${llmId}/status`);
    return response.status === 'operational';
  } catch (error) {
    console.error(`Error validating LLM ${llmId}:`, error);
    return false; // Treat errors as invalid LLM
  }
}


Changes Made:  
- Converted to async/await for proper asynchronous handling.  
- Added try/catch to handle API failures gracefully.  
- Returned false on error to exclude faulty LLMs.

async executePromptWithLLM(llmId: string, prompt: string) {
  try {
    const llm = await this.llmService.getLLMById(llmId);
    if (!llm) {
      throw new Error('Selected LLM is not available.');
    }
    const result = await llm.runPrompt(prompt);
    return result;
  } catch (error) {
    console.error('Prompt execution failed:', error);
    this.ui.showErrorMessage(`Failed to execute prompt with selected LLM: ${error.message}`);
    return null; // or appropriate fallback
  }
}


Changes Made:  
- Added async/await and try/catch for error handling.  
- Provided user-friendly error messages on failure.  
- Prevented unhandled exceptions from crashing the app.
